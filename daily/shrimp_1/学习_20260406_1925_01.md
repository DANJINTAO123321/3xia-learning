### 1. 注册文件模式 (sketchup-stl.rb)

```ruby
require 'sketchup'
require 'extensions'

module CommunityExtensions
  module STL
    # ★ 关键：处理Ruby 2.0在非英文用户名下的编码问题 ★
    current_path = File.dirname(__FILE__)
    if current_path.respond_to?(:force_encoding)
      current_path.force_encoding("UTF-8")
    end

    # 定义路径常量
    PLUGIN_ROOT_PATH    = current_path.freeze
    PLUGIN_PATH         = File.join(PLUGIN_ROOT_PATH, 'sketchup-stl').freeze
    PLUGIN_STRINGS_PATH = File.join(PLUGIN_PATH, 'strings').freeze

    # 加载国际化翻译器
    Sketchup::require File.join(PLUGIN_PATH, 'translator')
    options = {
      :custom_path => PLUGIN_STRINGS_PATH,
      :debug => false
    }
    @translator = Translator.new('STL.strings', options)

    # 翻译辅助方法（全局可访问）
    def self.translate(string)
      @translator.get(string)
    end

    # 创建扩展
    extension = SketchupExtension.new(
      STL.translate('STL Import & Export'),  # 本地化名称
      File.join(PLUGIN_PATH, 'loader.rb')    # 加载器
    )

    extension.description = STL.translate('Adds STL file format import...')
    extension.version = '2.2.0'
    extension.copyright = '2012-2017 Trimble Inc...'
    extension.creator = 'J. Foltz, N. Bromham, K. Shroeder, SketchUp Team'

    Sketchup.register_extension(extension, true)
  end
end
```

**关键经验：**
1. **编码兼容**：处理非ASCII用户名问题（Ruby 2.0 bug）
2. **路径常量**：使用freeze提升性能
3. **国际化**：所有用户可见文本通过translate()
4. **详细元信息**：版本、作者、版权完整

### 3. 导出器模式 (exporter.rb) - 重要！

```ruby
module CommunityExtensions
  module STL
    module Exporter

      # ★ SKUI嵌入到当前模块 ★
      load File.join(PLUGIN_PATH, 'SKUI', 'embed_skui.rb')
      ::SKUI.embed_in(self)

      # 默认选项
      OPTIONS = {
        'selection_only' => false,
        'export_units'   => 'Model Units',
        'stl_format'     => STL_ASCII
      }

      # 偏好设置键
      PREF_KEY = 'CommunityExtensions\\STL\\Exporter'.freeze

      # 文件扩展名
      def self.file_extension
        'stl'
      end

      # 主导出方法
      def self.export(path, export_entities, options = OPTIONS)
        # 处理ASCII/Binary格式
        if options['stl_format'] == STL_BINARY
          @write_face = method(:write_face_binary)
        else
          @write_face = method(:write_face_ascii)
        end
        
        scale = scale_factor(options['export_units'])
        write_header(file, model_name, options['stl_format'])
        facet_count = find_faces(file, export_entities, 0, scale, Geom::Transformation.new)
        write_footer(file, facet_count, model_name, options['stl_format'])
      end

      # 递归遍历所有面
      def self.find_faces(file, entities, facet_count, scale, tform)
        entities.each do |entity|
          next if entity.hidden? || !entity.layer.visible?
          
          if entity.is_a?(Sketchup::Face)
            facet_count += write_face(file, entity, scale, tform)
          elsif entity.is_a?(Sketchup::Group) ||
                entity.is_a?(Sketchup::ComponentInstance)
            # ★ 递归处理组和组件实例 ★
            entity_definition = Utils.definition(entity)
            facet_count += find_faces(
              file,
              entity_definition.entities,
              0,
              scale,
              tform * entity.transformation
            )
          end
        end
        facet_count
      end

      # 偏好设置读写封装
      def self.read_setting(key, default)
        Sketchup.read_default(PREF_KEY, key, default)
      end

      def self.write_setting(key, value)
        Sketchup.write_default(PREF_KEY, key, value)
      end

      # 单位换算
      def self.scale_factor(unit_key)
        case selected_key
        when 'Meters'       then 0.0254
        when 'Centimeters'  then 2.54
        when 'Millimeters'  then 25.4
        when 'Feet'         then 0.0833333333333333
        when 'Inches'       then 1.0
        end
      end
    end
  end
end
```

**核心设计模式：**

| 模式 | 说明 | 好处 |
|-----|------|------|
| **模块化Exporter** | 导出逻辑封装在module内 | 避免全局污染 |
| **递归遍历** | 处理Group/ComponentInstance | 支持嵌套结构 |
| **选项+持久化** | OPTIONS + read/write_setting | 用户设置持久化 |
| **SKUI嵌入** | ::SKUI.embed_in(self) | UI组件直接可用 |

### 5. 错误处理模式

```ruby
begin
  export(path, export_entities, OPTIONS) unless path.nil?
rescue => exception
  msg = "SketchUp STL Exporter:\n\n"
  msg << "An error occured during export.\n\n"
  msg << exception.message << "\n"
  msg << exception.backtrace.join("\n")
  UI.messagebox(msg, MB_MULTILINE)
end
```
