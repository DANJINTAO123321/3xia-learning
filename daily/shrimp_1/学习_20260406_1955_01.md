```ruby
def count_edges(entities)
  count = 0
  entities.each do |entity|
    if entity.is_a?(Sketchup::Edge)
      count += 1
    elsif entity.is_a?(Sketchup::Group) || 
          entity.is_a?(Sketchup::ComponentInstance)

end
```

### 详细模型分析报告

```ruby
def analyze_model
  model = Sketchup.active_model
  
  stats = {
    faces: 0,
    edges: 0,
    groups: 0,
    components: 0,
    materials: model.materials.length,
    layers: model.layers.length,
    images: 0,
    hidden_entities: 0
  }
  
  traverse_entities(model.entities, stats)
  
  puts "=== 模型分析报告 ==="
  puts "面数: #{stats[:faces]}"
  puts "边数: #{stats[:edges]}"
  puts "组数: #{stats[:groups]}"
  puts "组件数: #{stats[:components]}"
  puts "材质数: #{stats[:materials]}"
  puts "图层数: #{stats[:layers]}"
  puts "图片数: #{stats[:images]}"
  puts "隐藏实体: #{stats[:hidden_entities]}"
end

def traverse_entities(entities, stats)
  entities.each do |entity|
    case entity
    when Sketchup::Face
      stats[:faces] += 1
    when Sketchup::Edge
      stats[:edges] += 1
    when Sketchup::Group
      stats[:groups] += 1
      traverse_entities(entity.entities, stats)
    when Sketchup::ComponentInstance
      stats[:components] += 1
      traverse_entities(entity.definition.entities, stats)
    when Sketchup::Image
      stats[:images] += 1
    end
    
    if entity.hidden?
      stats[:hidden_entities] += 1
    end
  end
end
```

```ruby
def remove_hidden_entities
  model = Sketchup.active_model
  model.start_operation('Remove Hidden', true)
  
  hidden = []
  find_hidden(model.entities, hidden)
  
  if hidden.any?
    model.entities.erase_entities(hidden)
    puts "删除了 #{hidden.length} 个隐藏实体"
  end
  
  model.commit_operation
end

def find_hidden(entities, hidden)
  entities.each do |entity|
    if entity.hidden?
      hidden << entity
    end
    
    # 递归检查嵌套
    if entity.is_a?(Sketchup::Group)
      find_hidden(entity.entities, hidden)
    elsif entity.is_a?(Sketchup::ComponentInstance)
      find_hidden(entity.definition.entities, hidden)
    end
  end
end
```

```ruby
def cleanup_empty_groups
  model = Sketchup.active_model
  model.start_operation('Cleanup Empty Groups', true)
  
  removed = 0
  
  # 递归删除空组
  cleanup_recursive(model.entities)
  
  # 清理未使用的组件定义
  model.definitions.each do |defn|
    if defn.count_used_instances == 0 && !defn.group? && !defn.image?
      # 检查是否在选择中
      puts "未使用组件: #{defn.name}"
    end
  end
  
  model.commit_operation
  puts "清理完成，删除了 #{removed} 个空组"
end

def cleanup_recursive(entities)
  to_remove = []
  
  entities.each do |entity|
    if entity.is_a?(Sketchup::Group) && entity.entities.empty?
      to_remove << entity
    elsif entity.is_a?(Sketchup::Group)
      cleanup_recursive(entity.entities)
    elsif entity.is_a?(Sketchup::ComponentInstance)
      cleanup_recursive(entity.definition.entities)
    end
  end
  
  entities.erase_entities(to_remove) if to_remove.any?
end
```

### 策略4: 材质优化 - 合并重复材质

```ruby
def merge_duplicate_materials
  model = Sketchup.active_model
  materials = model.materials
  
  # 按颜色/纹理分组
  material_groups = {}
  
  materials.each do |mat|
    next unless mat.kind_of?(Sketchup::Material)
    key = material_key(mat)
    material_groups[key] ||= []
    material_groups[key] << mat
  end
  
  # 合并同组材质
  merged = 0
  material_groups.each do |key, mats|
    next if mats.length < 2
    primary = mats.first
    
    mats[1..-1].each do |mat|
      # 将所有使用副材质的面改用主材质
      remap_material_usages(mat, primary)
      materials.remove(mat)
      merged += 1
    end
  end
  
  puts "合并了 #{merged} 个重复材质"
end

def material_key(mat)
  # 根据材质属性生成唯一键
  color = mat.color ? mat.color.to_s : ""
  texture = mat.texture ? mat.texture.filename : ""
  "#{color}_#{texture}"
end

def remap_material_usages(from_mat, to_mat)
  model = Sketchup.active_model
  
  model.entities.each do |entity|
    remap_entity_material(entity, from_mat, to_mat)
  end
end

def remap_entity_material(entity, from_mat, to_mat)
  if entity.is_a?(Sketchup::Face)
    entity.material = to_mat if entity.material == from_mat
    entity.back_material = to_mat if entity.back_material == from_mat
  elsif entity.is_a?(Sketchup::Group)
    entity.entities.each { |e| remap_entity_material(e, from_mat, to_mat) }
  elsif entity.is_a?(Sketchup::ComponentInstance)
    entity.definition.entities.each { |e| remap_entity_material(e, from_mat, to_mat) }
  end
end
```

```ruby
def analyze_textures
  model = Sketchup.active_model
  
  total_texture_size = 0
  texture_count = 0

    next unless mat.texture
    
    texture = mat.texture

    puts "材质: #{mat.name}"
    puts "  纹理文件: #{texture.filename}"
    puts "  尺寸: #{texture.width} x #{texture.height} px"
    puts "  大小: #{size_kb.round(2)} KB"

    texture_count += 1

  puts "\n=== 纹理统计 ==="

end
```

### ⚠️ Ruby API限制

**重要：SketchUp Ruby API没有直接修改纹理尺寸的方法！**

解决方案：
1. **外部工具**：调用ImageMagick等命令行工具处理
2. **C++扩展**：用原生代码实现纹理缩放
3. **第三方库**：使用Ruby的RMagick库

```ruby
# 方案：调用ImageMagick压缩纹理
require 'open3'

def resize_texture(input_path, output_path, max_size = 1024)
  # 使用ImageMagick convert命令
  cmd = "convert #{input_path} -resize #{max_size}x#{max_size} #{output_path}"
  Open3.capture3(cmd)
end

# 处理所有纹理
model.materials.each do |mat|
  next unless mat.texture
  original = mat.texture.filename
  
  # 生成临时压缩文件
  compressed = original.gsub(/\.[^.]+$/, '_compressed$&')
  
  if resize_texture(original, compressed)
    # 替换材质纹理
    new_texture = mat.texture.copy
    new_texture.filename = compressed
  end
end
```

## 五、模型导出为轻量格式

### 导出选项

```ruby
# SketchUp Pro支持的导出格式
# .dae (COLLADA) - 推荐Web/移动端
# .obj - 通用3D格式
# .3ds - 传统游戏格式
# .kmz - Google Earth格式
# .glb / .usdz - SketchUp 2024+支持

def export_for_web(output_path)
  model = Sketchup.active_model
  
  # COLLADA导出选项
  options = {
    'geometry' => true,
    'texturemaps' => true,
    'triangulate' => true,  # 三角化（减少面）
    'export_hidden' => false,
    'units' => 'Meters'
  }
  
  # 导出为DAE（COLLADA）
  model.export(output_path + '.dae', options)
  
  # 导出为GLB（Draco压缩，更小）
  model.export(output_path + '.glb')
  
  puts "导出完成: #{output_path}"
end
```

## 六、批量处理框架

### 批量优化单个模型

```ruby
def optimize_model(options = {})
  defaults = {
    remove_hidden: true,
    remove_backfaces: true,
    merge_materials: true,
    purge_unused: true,
    target_texture_size: 1024  # 0 = 不缩放
  }
  opts = defaults.merge(options)
  
  model = Sketchup.active_model
  model.start_operation('Optimize Model', true)
  
  if opts[:remove_hidden]
    remove_hidden_entities
  end
  
  if opts[:remove_backfaces]
    remove_backfaces
  end
  
  if opts[:merge_materials]
    merge_duplicate_materials
  end
  
  if opts[:purge_unused]
    purge_unused_materials
    cleanup_empty_groups
  end
  
  model.commit_operation
  
  puts "模型优化完成！"
end
```

### 批量处理文件夹中的所有模型

```ruby
def batch_optimize(folder_path, options = {})
  require 'fileutils'
  
  Dir.glob("#{folder_path}/*.skp").each do |skp_file|
    puts "处理: #{skp_file}"
    
    # 打开模型
    model = Sketchup.open_file(skp_file)
    next unless model
    
    # 优化
    optimize_model(options)
    
    # 保存
    model.save
    model.close
    
    puts "完成: #{skp_file}"
  end
end
```
