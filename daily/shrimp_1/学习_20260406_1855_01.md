### 创建基本图元

#### 创建直线 (Line)
```python
from pyautocad import Autocad
from pyautocad.util import points_to_array

acad = Autocad()

# 起点(0,0,0) 到 终点(1000, 1000, 0)
start_point = (0, 0, 0)
end_point = (1000, 1000, 0)

line = acad.model.AddLine(points_to_array([start_point, end_point]))
print(f'创建直线: {line}')
```

#### 创建圆 (Circle)
```python
# 圆心(0, 0, 0)，半径500
center_point = (0, 0, 0)
radius = 500

circle = acad.model.AddCircle(points_to_array([center_point]), radius)
print(f'创建圆: {circle}')
```

#### 创建圆弧 (Arc)
```python
# 圆心(0, 0, 0)，半径500，起始角度0°，结束角度90°
center = (0, 0, 0)
radius = 500
start_angle = 0  # 弧度
end_angle = Math.pi / 2  # 90°

arc = acad.model.AddArc(points_to_array([center]), radius, start_angle, end_angle)
```

#### 切换当前图层
```python
def switch_to_layer(acad, layer_name):
    """切换到指定图层"""
    if layer_name in acad.doc.layers:
        acad.doc.ActiveLayer = acad.doc.layers[layer_name]
        return True
    return False

# 使用
switch_to_layer(acad, 'DIM')
```

#### 创建新图层
```python
def ensure_layer(acad, layer_name, color=7, linetype='Continuous'):
    """确保图层存在，不存在则创建"""
    if layer_name not in acad.doc.layers:
        layer = acad.doc.layers.add(layer_name)
        layer.color = color
        if linetype in acad.doc.linetypes:
            layer.linetype = linetype
    acad.doc.ActiveLayer = acad.doc.layers[layer_name]
    return acad.doc.ActiveLayer

# 使用
ensure_layer(acad, '我的图层', color=1)  # 红色
```

### 最简单的插件模板

#### my_first_extension.rb
```ruby
# 文件: my_first_extension.rb
# 位置: SketchUp插件目录

require 'sketchup'

# 定义扩展信息
module MyFirstExtension
  VERSION = "1.0.0"
  
  # 插件加载时自动执行
  def self.startup
    puts "MyFirstExtension v#{VERSION} 已加载!"
    
    # 创建菜单项
    add_menu_items
  end
  
  # 添加菜单项
  def self.add_menu_items
    # 添加到"扩展"菜单
    extensions_menu = UI.menu("Extensions")
    extensions_menu.add_item("我的第一个插件") {
      # 这里放置插件功能
      Sketchup.active_model.start_operation("显示消息")
      UI.messagebox("Hello from SketchUp Ruby!")
      Sketchup.active_model.commit_operation
    }
  end
end

# 启动插件
MyFirstExtension.startup
```

### 创建几何体

#### 创建立方体
```ruby
# 获取当前模型和起点
model = Sketchup.active_model
entities = model.entities

# 起点坐标
origin = Geom::Point3d.new(0, 0, 0)

# 定义立方体的6个面 (每个面4个顶点)
width = 100
depth = 100
height = 100

# 创建立方体组
group = entities.add_group

# 添加立方体的面
pts = [
  Geom::Point3d.new(0, 0, 0),
  Geom::Point3d.new(width, 0, 0),
  Geom::Point3d.new(width, depth, 0),
  Geom::Point3d.new(0, depth, 0)
]

# 正面
face1 = group.entities.add_face(pts)

# 拉伸 (pushpull) 100单位
face1.pushpull(height)

puts "立方体已创建: #{width}x#{depth}x#{height}"
```

#### 创建圆柱体
```ruby
model = Sketchup.active_model
entities = model.entities

# 圆心
center = Geom::Point3d.new(0, 0, 0)
radius = 50
sides = 24  # 边的数量（越多越圆）

# 创建圆
circle = entities.add_circle(center, [0, 0, 1], radius, sides)

# 获取圆产生的面
face = entities.grep(Sketchup::Face).last

# 拉伸
face.pushpull(100)

puts "圆柱体已创建: 半径=#{radius}, 高度=100"
```

### 与AutoCAD的差异

| 功能 | AutoCAD (Python) | SketchUp (Ruby) |
|------|-----------------|-----------------|
| 连接方式 | ActiveX COM | 内嵌解释器 |
| 创建图元 | AddLine/AddCircle | add_circle/add_group |
| 拉伸 | 需用region | pushpull方法 |
| 选择 | selection.each | selection.grep |
| 坐标系统 | (x,y,z) | Geom::Point3d |
