## pyautocad 实战脚本研究

### 1. 切换图层的正确方法

`python
from pyautocad import Autocad

acad = Autocad()

# 列出所有图层
for layer in acad.doc.layers:
    print(f"图层: {layer.name}, 颜色: {layer.color}")

# 切换到指定图层
def switch_to_layer(acad, layer_name):
    if layer_name in acad.doc.layers:
        acad.doc.ActiveLayer = acad.doc.layers[layer_name]
        print(f'已切换到图层: {layer_name}')
        return True
    else:
        print(f'图层不存在: {layer_name}')
        return False

# 使用示例
switch_to_layer(acad, '0')
switch_to_layer(acad, 'DIM')
`

### 2. 遍历图层找匹配项

`python
def find_layer_by_pattern(acad, pattern):
    ''搜索包含关键词的图层''
    matching = []
    for layer in acad.doc.layers:
        if pattern.lower() in layer.name.lower():
            matching.append(layer.name)
    return matching

# 使用示例
dims = find_layer_by_pattern(acad, 'DIM')
print(f'找到的标注图层: {dims}')
`

### 3. 创建图层快捷函数

`python
def ensure_layer(acad, layer_name, color=7, linetype='Continuous'):
    ''确保图层存在，不存在则创建''
    if layer_name not in acad.doc.layers:
        layer = acad.doc.layers.add(layer_name)
        layer.color = color
        if linetype in acad.doc.linetypes:
            layer.linetype = linetype
    acad.doc.ActiveLayer = acad.doc.layers[layer_name]
    return acad.doc.ActiveLayer

# 使用示例
ensure_layer(acad, '我的图层', color=1)  # 红色
`

### 4. 获取当前图层信息

`python
def get_current_layer_info(acad):
    layer = acad.doc.ActiveLayer
    return {
        'name': layer.name,
        'color': layer.color,
        'linetype': layer.linetype,
        'locked': layer.lock,
        'visible': layer.on
    }
`

### 5. 隐藏/显示图层

`python
def toggle_layer_visibility(acad, layer_name, visible=True):
    if layer_name in acad.doc.layers:
        acad.doc.layers[layer_name].on = visible

def freeze_layer(acad, layer_name):
    if layer_name in acad.doc.layers:
        acad.doc.layers[layer_name].freeze = True
`
